- [ ] **Step 2** — 新建 `analysis/inference.py`。DeLong 用 Sun & Xu (2014) 的 midrank 快速实现，**不得**依赖 statsmodels 或任何未安装的包：

```python
import numpy as np
import pandas as pd
from scipy.stats import norm
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score, roc_auc_score

from analysis.config import RANDOM_SEED
from analysis.evaluation import _holm, youden_sens_spec


def _midrank(x):
    """1-based midrank，并列值取平均秩。"""
    x = np.asarray(x, dtype=float)
    order = np.argsort(x, kind="mergesort")
    xs = x[order]
    n = len(x)
    r = np.empty(n, dtype=float)
    i = 0
    while i < n:
        j = i
        while j < n - 1 and xs[j + 1] == xs[i]:
            j += 1
        r[i:j + 1] = 0.5 * (i + j) + 1.0
        i = j + 1
    out = np.empty(n, dtype=float)
    out[order] = r
    return out


def delong_roc_variance(y_true, y_scores):
    """y_scores: (k, n)。返回 (aucs (k,), cov (k, k))。"""
    y = np.asarray(y_true).astype(int)
    S = np.atleast_2d(np.asarray(y_scores, dtype=float))
    pos = S[:, y == 1]
    neg = S[:, y == 0]
    m, n = pos.shape[1], neg.shape[1]
    k = S.shape[0]
    tx = np.empty((k, m)); ty = np.empty((k, n)); tz = np.empty((k, m + n))
    for r in range(k):
        pr, nr = pos[r], neg[r]
        tx[r] = _midrank(pr)
        ty[r] = _midrank(nr)
        tz[r] = _midrank(np.concatenate([pr, nr]))
    aucs = tz[:, :m].sum(axis=1) / (m * n) - (m + 1.0) / (2.0 * n)
    # Sun & Xu (2014) 的结构分量必须按 n / m 归一化，否则方差被放大 O(m*n) 倍。
    # 注意变量名沿用文献习惯但方向相反：v_pos 是阳性侧分量（文献记 V10），
    # v_neg 是阴性侧分量（文献记 V01）；Var(AUC) = S_pos/m + S_neg/n。
    v_pos = (tz[:, :m] - tx) / n
    v_neg = 1.0 - (tz[:, m:] - ty) / m
    s_pos = np.atleast_2d(np.cov(v_pos) if k > 1 else np.var(v_pos[0], ddof=1))
    s_neg = np.atleast_2d(np.cov(v_neg) if k > 1 else np.var(v_neg[0], ddof=1))
    return aucs, s_pos / m + s_neg / n


def delong_test(y_true, prob_a, prob_b):
    """双侧 DeLong 检验，H0：两模型 AUC 相等。返回 (auc_a, auc_b, z, p)。"""
    aucs, cov = delong_roc_variance(y_true, np.vstack([prob_a, prob_b]))
    var = cov[0, 0] + cov[1, 1] - 2 * cov[0, 1]
    if var <= 0:
        return float(aucs[0]), float(aucs[1]), 0.0, 1.0
    z = float((aucs[0] - aucs[1]) / np.sqrt(var))
    return float(aucs[0]), float(aucs[1]), z, float(2 * norm.sf(abs(z)))


def delong_ci(y_true, prob, alpha=0.05):
    """AUC 的 DeLong 解析置信区间。返回 (auc, lo, hi)。"""
    aucs, cov = delong_roc_variance(y_true, np.asarray(prob, dtype=float).reshape(1, -1))
    se = float(np.sqrt(max(cov[0, 0], 0.0)))
    z = norm.ppf(1 - alpha / 2)
    return float(aucs[0]), float(aucs[0] - z * se), float(aucs[0] + z * se)


def bootstrap_auc_ci(y_true, y_prob, n_boot=2000, alpha=0.05, seed=RANDOM_SEED):
    """百分位 bootstrap AUC 置信区间。返回 (auc, lo, hi, boots)。"""
    y = np.asarray(y_true).astype(int); p = np.asarray(y_prob, dtype=float)
    point = float(roc_auc_score(y, p))
    rng = np.random.default_rng(seed)
    n = len(y)
    boots = []
    for _ in range(n_boot):
        idx = rng.integers(0, n, n)
        if len(np.unique(y[idx])) < 2:
            continue
        boots.append(roc_auc_score(y[idx], p[idx]))
    boots = np.asarray(boots, dtype=float)
    if boots.size == 0:
        return point, point, point, boots
    return (point, float(np.percentile(boots, 100 * alpha / 2)),
            float(np.percentile(boots, 100 * (1 - alpha / 2))), boots)


def calibration_intercept_slope(y_true, y_prob):
    """在 logit(p) 上重新拟合 logistic，返回 (截距, 斜率)。理想校准为 (0, 1)。"""
    p = np.clip(np.asarray(y_prob, dtype=float), 1e-6, 1 - 1e-6)
    z = np.log(p / (1 - p)).reshape(-1, 1)
    lr = LogisticRegression(C=1e6, max_iter=10000, tol=1e-10).fit(z, np.asarray(y_true))
    return float(lr.intercept_[0]), float(lr.coef_[0, 0])


def pooled_oof(y, oof_records):
    """把 [(test_index, proba), ...] 拼回与原样本对齐的 OOF 概率向量；
    多次重复覆盖同一样本时取均值（重复 CV 的 averaged OOF）。"""
    total = np.zeros(len(y)); count = np.zeros(len(y))
    for idx, proba in oof_records:
        total[idx] += proba
        count[idx] += 1
    if not np.all(count > 0):
        raise ValueError("OOF predictions must cover all samples")
    return total / count


def pooled_performance(y, oof_by_model: dict, n_boot=2000, seed=RANDOM_SEED) -> pd.DataFrame:
    """每个模型在 pooled OOF 上的完整性能表。"""
    y = np.asarray(y).astype(int)
    rows = []
    for name, recs in oof_by_model.items():
        p = pooled_oof(y, recs)
        auc, lo, hi, _ = bootstrap_auc_ci(y, p, n_boot=n_boot, seed=seed)
        _, dlo, dhi = delong_ci(y, p)
        intercept, slope = calibration_intercept_slope(y, p)
        sens, spec = youden_sens_spec(y, p)
        rows.append({"model": name, "n": int(len(y)), "events": int(y.sum()),
                     "auc": auc, "auc_ci_boot_lo": lo, "auc_ci_boot_hi": hi,
                     "auc_ci_delong_lo": dlo, "auc_ci_delong_hi": dhi,
                     "auprc": float(average_precision_score(y, p)),
                     "brier": float(np.mean((p - y) ** 2)),
                     "calib_intercept": intercept, "calib_slope": slope,
                     "sensitivity": sens, "specificity": spec})
    return (pd.DataFrame(rows).sort_values("auc", ascending=False)
            .reset_index(drop=True))


def delong_pairwise(y, oof_by_model: dict,
                    baseline="Logistic_LASSO_baseline") -> pd.DataFrame:
    """pooled OOF 上每个模型 vs 基线的 DeLong 检验 + Holm 校正。"""
    y = np.asarray(y).astype(int)
    pooled = {k: pooled_oof(y, v) for k, v in oof_by_model.items()}
    if baseline not in pooled:
        raise KeyError(f"baseline '{baseline}' not in {sorted(pooled)}")
    rows = []
    for name, p in pooled.items():
        if name == baseline:
            continue
        auc_a, auc_b, z, pv = delong_test(y, p, pooled[baseline])
        rows.append({"model": name, "auc": auc_a, "auc_baseline": auc_b,
                     "auc_diff": auc_a - auc_b, "z": z, "p_delong": pv})
    out = pd.DataFrame(rows)
    if len(out):
        out["p_holm"] = _holm(out["p_delong"].to_numpy())
    return out.sort_values("auc_diff", ascending=False).reset_index(drop=True)
```

def column_kinds(df) -> dict:
    """返回 {'binary': [...], 'continuous': [...]}；判定标准与 AutoPreprocessor 一致：
    取值 ⊆ {0,1}（忽略 NaN）视为二值。不含 TARGET_COL。"""
    feats = [c for c in df.columns if c != TARGET_COL]
    binary, cont = [], []
    for c in feats:
        v = df[c].dropna().unique()
        (binary if set(v) <= {0, 1} else cont).append(c)
    return {"binary": binary, "continuous": cont}

```python
import numpy as np
import pandas as pd
from scipy.stats import fisher_exact, mannwhitneyu, norm

from analysis.config import TARGET_COL
from analysis.data_loader import column_kinds


def irls_logistic(X, y, max_iter=100, tol=1e-10):
    """带截距的无惩罚 logistic 回归（Newton–Raphson / IRLS）。

    返回 dict(coef=[截距, *斜率], se, z, p, ci_lo, ci_hi)，同一尺度。
    Fisher 信息奇异或权重退化（完全分离）时抛 ValueError。
    """
    X = np.asarray(X, dtype=float)
    y = np.asarray(y, dtype=float).ravel()
    n, k = X.shape
    A = np.column_stack([np.ones(n), X])
    ybar = float(np.clip(y.mean(), 1e-9, 1 - 1e-9))
    beta = np.zeros(k + 1)
    beta[0] = np.log(ybar / (1 - ybar))
    converged = False
    for _ in range(max_iter):
        mu = 1.0 / (1.0 + np.exp(-np.clip(A @ beta, -30, 30)))
        W = mu * (1 - mu)
        if np.any(W <= 1e-12):
            raise ValueError("degenerate weights (perfect separation?)")
        grad = A.T @ (y - mu)
        H = A.T @ (A * W[:, None]) + 1e-10 * np.eye(k + 1)
        try:
            step = np.linalg.solve(H, grad)
        except np.linalg.LinAlgError as e:
            raise ValueError(f"singular Fisher information: {e}") from e
        beta = beta + step
        if np.max(np.abs(step)) < tol:
            converged = True
            break
    if not converged:
        raise ValueError("IRLS did not converge")
    mu = 1.0 / (1.0 + np.exp(-np.clip(A @ beta, -30, 30)))
    W = mu * (1 - mu)
    cov = np.linalg.inv(A.T @ (A * W[:, None]) + 1e-10 * np.eye(k + 1))
    se = np.sqrt(np.clip(np.diag(cov), 0, None))
    z = np.divide(beta, se, out=np.zeros_like(beta), where=se > 0)
    return {"coef": beta, "se": se, "z": z, "p": 2 * norm.sf(np.abs(z)),
            "ci_lo": beta - norm.ppf(0.975) * se,
            "ci_hi": beta + norm.ppf(0.975) * se}


def univariate_lr_table(df, feature_names=None, p_threshold=0.05) -> pd.DataFrame:
    """复现已发表论文的单变量 logistic 筛选（Table 2）。

    连续变量按 1 个标准差报告 OR，二值变量直接报告 OR。
    不可估计的变量记 NaN 并在 note 列说明，不中断整表。
    """
    df = df.reset_index(drop=True)
    names = feature_names or [c for c in df.columns if c != TARGET_COL]
    y = df[TARGET_COL].to_numpy(dtype=float)
    kinds = column_kinds(df)
    rows = []
    for c in names:
        x = df[c].to_numpy(dtype=float)
        sd = float(np.nanstd(x))
        if sd == 0 or np.isnan(sd):
            rows.append({"variable": c, "n": int(len(x)), "or_per_sd": np.nan,
                         "ci_lo": np.nan, "ci_hi": np.nan, "p_wald": np.nan,
                         "selected": False, "note": "constant column"})
            continue
        is_binary = c in kinds["binary"]
        xs = x if is_binary else (x - float(np.nanmean(x))) / sd
        xs = np.where(np.isnan(xs), np.nanmedian(xs), xs)
        try:
            fit = irls_logistic(xs.reshape(-1, 1), y)
            rows.append({"variable": c, "n": int(len(x)),
                         "or_per_sd": float(np.exp(fit["coef"][1])),
                         "ci_lo": float(np.exp(fit["ci_lo"][1])),
                         "ci_hi": float(np.exp(fit["ci_hi"][1])),
                         "p_wald": float(fit["p"][1]),
                         "selected": bool(fit["p"][1] < p_threshold),
                         "note": "binary (OR)" if is_binary else "OR per 1 SD"})
        except ValueError as e:
            rows.append({"variable": c, "n": int(len(x)), "or_per_sd": np.nan,
                         "ci_lo": np.nan, "ci_hi": np.nan, "p_wald": np.nan,
                         "selected": False, "note": f"not estimable: {e}"})
    return (pd.DataFrame(rows).sort_values("p_wald", na_position="last")
            .reset_index(drop=True))


def _median_iqr(s) -> str:
    s = pd.Series(s).dropna()
    if s.empty:
        return "NA"
    q1, med, q3 = s.quantile([0.25, 0.5, 0.75])
    return f"{med:.2f} ({q1:.2f}-{q3:.2f})"


def _smd(a, b) -> float:
    a = pd.Series(a).dropna().to_numpy(dtype=float)
    b = pd.Series(b).dropna().to_numpy(dtype=float)
    if a.size < 2 or b.size < 2:
        return np.nan
    pooled = np.sqrt((a.var(ddof=1) + b.var(ddof=1)) / 2)
    return float((a.mean() - b.mean()) / pooled) if pooled > 0 else 0.0


def table1_by_group(df, group_col=TARGET_COL) -> pd.DataFrame:
    """稿件 Table 1：应答 vs 非应答的基线特征比较。

    连续变量：中位数 (IQR) + Mann–Whitney U；二值变量：n (%) + Fisher 精确检验。
    另附标准化均差 SMD（|SMD| > 0.1 视为有意义的组间差异）。
    """
    df = df.reset_index(drop=True)
    kinds = column_kinds(df)
    g1, g0 = df[df[group_col] == 1], df[df[group_col] == 0]
    rows = []
    for c in [x for x in df.columns if x != group_col]:
        v = df[c]
        if c in kinds["binary"]:
            n1, n0 = int(g1[c].sum()), int(g0[c].sum())
            try:
                _, p = fisher_exact([[n1, len(g1) - n1], [n0, len(g0) - n0]])
            except ValueError:
                p = np.nan
            rows.append({"variable": c, "type": "binary",
                         "overall": f"{int(v.sum())}/{len(v)} ({v.mean() * 100:.1f}%)",
                         "responder": f"{n1}/{len(g1)} ({n1 / len(g1) * 100:.1f}%)",
                         "non_responder": f"{n0}/{len(g0)} ({n0 / len(g0) * 100:.1f}%)",
                         "p_value": float(p), "smd": _smd(g1[c], g0[c])})
        else:
            a, b = g1[c].dropna(), g0[c].dropna()
            _, p = mannwhitneyu(a, b, alternative="two-sided") if a.size and b.size \
                else (np.nan, np.nan)
            rows.append({"variable": c, "type": "continuous", "overall": _median_iqr(v),
                         "responder": _median_iqr(g1[c]),
                         "non_responder": _median_iqr(g0[c]),
                         "p_value": float(p), "smd": _smd(g1[c], g0[c])})
    return pd.DataFrame(rows)
```

def make_stacking_pipeline(seed=RANDOM_SEED, light=False) -> Pipeline:
    """二层概率 stacking：5 个异质基学习器 + L2 logistic 元学习器。

    基学习器超参固定（不调），只调元学习器的 C，以控制嵌套 CV 的计算量。
    最外层是 AutoPreprocessor，基学习器看到的都是折内预处理后的数据；
    LASSO 基学习器内部仍带 UnivariateLRFilter，选择发生在折内。
    """
    lasso = Pipeline([
        ("univar", UnivariateLRFilter(p_threshold=0.05)),
        ("clf", LogisticRegression(penalty="l1", solver="liblinear", C=1.0,
                                   max_iter=5000, random_state=seed))])
    bases = [
        ("lasso", lasso),
        ("svm", SVC(probability=True, C=1.0, gamma="scale", random_state=seed)),
        ("rf", RandomForestClassifier(n_estimators=30 if light else 200,
                                      max_features="sqrt", random_state=seed, n_jobs=1)),
        ("xgb", XGBClassifier(n_estimators=10 if light else 100, max_depth=2,
                              learning_rate=0.1, subsample=0.8, reg_lambda=1.0,
                              eval_metric="logloss", random_state=seed,
                              n_jobs=1, verbosity=0)),
        ("lgbm", LGBMClassifier(n_estimators=10 if light else 100, num_leaves=7,
                                min_child_samples=10, random_state=seed,
                                n_jobs=1, verbose=-1)),
    ]
    meta = LogisticRegression(penalty="l2", C=1.0, max_iter=5000, random_state=seed)
    stack = StackingClassifier(estimators=bases, final_estimator=meta,
                               cv=2 if light else 3, stack_method="predict_proba",
                               passthrough=False, n_jobs=1)
    return Pipeline([("pre", AutoPreprocessor()), ("clf", stack)])

### Task D: 表观模型复现与乐观度校正（`analysis/apparent.py`）

**理由:** 已发表论文报告表观 AUC 0.845、bootstrap 内部验证 0.814。本文核心叙事是"表观估计的乐观偏倚 vs 嵌套 CV 的诚实估计（0.746）"。必须复现表观协议并给出 Efron–Gong 乐观度校正，才能把这条对照写成可辩护的结果而非口头断言。

- [ ] **Step 1** — 新建 `analysis/apparent.py`：

```python
import numpy as np
import pandas as pd
from sklearn.base import clone
from sklearn.metrics import roc_auc_score
from sklearn.model_selection import StratifiedKFold

from analysis.baseline import make_baseline_pipeline
from analysis.config import N_INNER_FOLDS, RANDOM_SEED
from analysis.evaluation import calibration_slope, youden_sens_spec

C_GRID = (0.01, 0.05, 0.1, 0.5, 1.0, 5.0, 10.0)


def fit_apparent(X, y, seed=RANDOM_SEED):
    """在**全样本**上复现已发表协议：单变量筛选（P<0.05）+ L1 logistic，
    C 由全样本内层 CV 选定。这是表观（apparent）模型，只用于展示乐观偏倚，
    **不得**作为验证性能报告。返回 dict。"""
    X = np.asarray(X, dtype=float); y = np.asarray(y).astype(int)
    pipe = make_baseline_pipeline(seed)
    cv = StratifiedKFold(N_INNER_FOLDS, shuffle=True, random_state=seed)
    best_C, best_auc = 1.0, -np.inf
    for C in C_GRID:
        aucs = []
        for tr, te in cv.split(X, y):
            cand = clone(pipe).set_params(clf__C=C).fit(X[tr], y[tr])
            aucs.append(roc_auc_score(y[te], cand.predict_proba(X[te])[:, 1]))
        m = float(np.mean(aucs))
        if m > best_auc:
            best_C, best_auc = C, m
    fitted = clone(pipe).set_params(clf__C=best_C).fit(X, y)
    p = fitted.predict_proba(X)[:, 1]
    sens, spec = youden_sens_spec(y, p)
    filt, clf = fitted.named_steps["univar"], fitted.named_steps["clf"]
    return {"pipeline": fitted, "C": best_C, "inner_cv_auc": best_auc,
            "apparent_auc": float(roc_auc_score(y, p)),
            "apparent_sensitivity": sens, "apparent_specificity": spec,
            "apparent_brier": float(np.mean((p - y) ** 2)),
            "apparent_calibration_slope": calibration_slope(y, p),
            "n_kept_after_univariate": int(len(filt.kept_idx_)),
            "kept_idx": list(filt.kept_idx_),
            "lasso_coefficients": pd.DataFrame({"coef": clf.coef_.ravel()}),
            "n_nonzero": int(np.sum(np.abs(clf.coef_.ravel()) > 1e-8)),
            "predictions": p}


def optimism_bootstrap(X, y, n_boot=200, seed=RANDOM_SEED, fit_fn=None):
    """Efron–Gong 乐观度 bootstrap：
    optimism = mean_b[ AUC(model_b on boot_b) - AUC(model_b on original) ]，
    corrected = apparent_full - optimism。区间由 optimism 分布的 2.5/97.5 百分位翻转得到。"""
    X = np.asarray(X, dtype=float); y = np.asarray(y).astype(int)
    fit_fn = fit_fn or (lambda Xb, yb: make_baseline_pipeline(seed).fit(Xb, yb))
    apparent = float(roc_auc_score(y, fit_fn(X, y).predict_proba(X)[:, 1]))
    rng = np.random.default_rng(seed)
    n = len(y)
    opt = []
    for _ in range(n_boot):
        idx = rng.integers(0, n, n)
        if len(np.unique(y[idx])) < 2:
            continue
        try:
            m = fit_fn(X[idx], y[idx])
            a_boot = float(roc_auc_score(y[idx], m.predict_proba(X[idx])[:, 1]))
            a_orig = float(roc_auc_score(y, m.predict_proba(X)[:, 1]))
        except (ValueError, RuntimeError):
            continue
        opt.append(a_boot - a_orig)
    opt = np.asarray(opt, dtype=float)
    if opt.size == 0:
        return {"apparent_auc": apparent, "optimism": np.nan, "optimism_sd": np.nan,
                "corrected_auc": np.nan, "corrected_ci_lo": np.nan,
                "corrected_ci_hi": np.nan, "n_boot_used": 0,
                "n_boot_requested": int(n_boot)}
    corrected = apparent - float(opt.mean())
    return {"apparent_auc": apparent, "optimism": float(opt.mean()),
            "optimism_sd": float(opt.std(ddof=1)) if opt.size > 1 else np.nan,
            "corrected_auc": corrected,
            "corrected_ci_lo": corrected - float(np.percentile(opt, 97.5)),
            "corrected_ci_hi": corrected - float(np.percentile(opt, 2.5)),
            "n_boot_used": int(opt.size), "n_boot_requested": int(n_boot)}
```

import numpy as np
import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline

from analysis.config import (N_FOLDS, N_INNER_FOLDS, N_REPEATS, RANDOM_SEED,
                             TARGET_COL)
from analysis.evaluation import run_comparison, summarize
from analysis.models import get_models
from analysis.preprocessing import AutoPreprocessor

# TabNet 与 MLP 排除理由：TabNet 在重复 CV 下最慢（GPU）且已显著劣于基线；
# MLP 已显著劣于基线。敏感性分析只检验**排序是否稳健**，不需要最慢的两个模型。
SENSITIVITY_MODELS = ("Logistic_LASSO_baseline", "Logistic_L2", "SVM_rbf",
                      "RF", "XGBoost", "LightGBM")


def build_scenarios(df):
    """返回 {scenario: (X, y, feature_names, models)}。不修改入参 df，不写 data.xlsx。"""
    feats = [c for c in df.columns if c != TARGET_COL]
    y = df[TARGET_COL].to_numpy(dtype=np.int64)
    X = df[feats].to_numpy(dtype=np.float64)
    all_models = get_models(seed=RANDOM_SEED, light=True)
    sub = {k: all_models[k] for k in SENSITIVITY_MODELS}
    out = {"primary": (X, y, feats, sub)}

    no_filter = dict(sub)
    no_filter["Logistic_LASSO_baseline"] = Pipeline([
        ("pre", AutoPreprocessor()),
        ("clf", LogisticRegression(penalty="l1", solver="liblinear", C=1.0,
                                   max_iter=5000, random_state=RANDOM_SEED))])
    out["no_univariate_filter"] = (X, y, feats, no_filter)

    keep = [c for c in feats if c != "Age"]
    out["drop_age"] = (df[keep].to_numpy(dtype=np.float64), y, keep, sub)

    cc = df.dropna()
    out["complete_case"] = (cc[feats].to_numpy(dtype=np.float64),
                            cc[TARGET_COL].to_numpy(dtype=np.int64), feats, sub)
    return out


def run_sensitivity(df, n_folds=N_FOLDS, n_repeats=N_REPEATS,
                    inner_folds=N_INNER_FOLDS, seed=RANDOM_SEED, scenarios=None):
    """逐情景跑嵌套 CV（不做网格调参以控制耗时），返回 tidy DataFrame：
    scenario, model, n_rows, n_features, mean_auc, sd_auc, delta_vs_primary。"""
    scenarios = scenarios or build_scenarios(df)
    frames = []
    for name, (X, y, feats, models) in scenarios.items():
        fold_df, _ = run_comparison(X, y, models, grids={}, n_folds=n_folds,
                                    n_repeats=n_repeats, inner_folds=inner_folds,
                                    seed=seed)
        for m, row in summarize(fold_df).set_index("model").iterrows():
            frames.append({"scenario": name, "model": m, "n_rows": int(len(y)),
                           "n_features": int(len(feats)),
                           "mean_auc": float(row["mean_auc"]),
                           "sd_auc": float(row["sd_auc"])})
    out = pd.DataFrame(frames)
    base = out[out.scenario == "primary"].set_index("model")["mean_auc"]
    out["delta_vs_primary"] = [
        r["mean_auc"] - base[r["model"]] if r["model"] in base.index else np.nan
        for _, r in out.iterrows()]
    return (out.sort_values(["scenario", "mean_auc"], ascending=[True, False])
            .reset_index(drop=True))


def ranking_stable(sens_df, top_model="SVM_rbf") -> dict:
    """报告结论稳健性：每个情景的第一名是谁、top_model 是否仍是第一。"""
    out = {}
    for sc, g in sens_df.groupby("scenario"):
        g = g.sort_values("mean_auc", ascending=False)
        out[sc] = {"winner": str(g.iloc[0]["model"]),
                   "winner_auc": float(g.iloc[0]["mean_auc"]),
                   "target_still_wins": bool(g.iloc[0]["model"] == top_model)}
    return out

```python
import json
import platform
from datetime import date
from pathlib import Path

import joblib
import numpy as np


def _lib_versions() -> dict:
    import numpy, pandas, scipy, sklearn
    d = {"numpy": numpy.__version__, "pandas": pandas.__version__,
         "scipy": scipy.__version__, "scikit-learn": sklearn.__version__}
    for name in ("xgboost", "lightgbm", "shap", "torch"):
        try:
            d[name] = __import__(name).__version__
        except Exception:
            pass
    return d


def _enc(o):
    if isinstance(o, np.integer):
        return int(o)
    if isinstance(o, np.floating):
        return float(o)
    if isinstance(o, np.ndarray):
        return o.tolist()
    raise TypeError(f"not JSON serializable: {type(o)}")


def freeze_model(estimator, feature_names, out_dir, meta=None):
    """持久化最优模型 + TRIPOD+AI 风格 model card。返回 (model_path, card_path)。"""
    out_dir = Path(out_dir); out_dir.mkdir(parents=True, exist_ok=True)
    model_path = out_dir / "best_model.joblib"
    joblib.dump({"estimator": estimator, "feature_names": list(feature_names)}, model_path)
    card = {"schema": "TRIPOD+AI model card", "version": 1,
            "created": date.today().isoformat(),
            "target": "CRT response (binary)",
            "n_features": len(feature_names), "feature_names": list(feature_names),
            "python": platform.python_version(), "libraries": _lib_versions(),
            **(meta or {})}
    card_path = out_dir / "model_card.json"
    card_path.write_text(json.dumps(card, indent=2, ensure_ascii=False, default=_enc),
                         encoding="utf-8")
    return str(model_path), str(card_path)


def load_model(model_path):
    """返回 (estimator, feature_names)。"""
    obj = joblib.load(model_path)
    return obj["estimator"], obj["feature_names"]
```

- [ ] **Step 1** — 新建 `analysis/app.py`，实现：
  - `build_bundle(estimator, feature_names, meta=None, shap_top=None) -> dict`：把模型与元数据打成 bundle（不写盘）。
  - `load_bundle(model_path) -> dict`：用 `freeze.load_model` 从 joblib 重建 bundle。
  - `serve(bundle, host="127.0.0.1", port=0) -> tuple[ThreadingHTTPServer, threading.Thread, int]`：启动守护线程，返回 `(server, thread, actual_port)`；`port=0` 由 OS 分配，供测试使用。
  - `GET /` → 200，`Content-Type: text/html; charset=utf-8`，含 `<form method="post" action="/predict">`，每个特征一个 `<input type="number" step="any" name="...">`，label 用 `explain.DISPLAY_NAMES` 的友好名。特征名必须经 `html.escape`。
  - `GET /health` → 200 JSON `{"status": "ok", "n_features": k}`。
  - `POST /predict` → 接受 `application/json`（`{"features": {name: value}}`）或 `application/x-www-form-urlencoded`；按 `feature_names` 顺序组装 `(1, k)` float 矩阵；缺失键 / 非数值 / 特征数不匹配 → **400** 且返回 JSON `{"error": "..."}`；成功 → 200 JSON `{"probability": p, "class": 0|1, "prediction": "responder"|"non-responder"}`，若 bundle 带 `shap_top` 再加 `"top_features": [...]`。
  - 其它路径 → 404 JSON。**禁止** `eval`/`exec`、禁止拼接 shell、禁止访问外网；所有异常都要捕获并转成 500 JSON，绝不让线程崩掉。
